99 /**
100 * Displays a formatted string on the emulated three-button LCD screen.
101 *
102 * This function uses the following values of errno when an error state is
103 * reached:
104 * ENXIO - The LCD has not been initialized. Call lcd_initialize() first.
105 * EINVAL - The line number specified is not in the range [0-7]
106 *
107 * \param line
108 * The line on which to display the text [0-7]
109 * \param fmt
110 * Format string
111 * \param ...args
112 * Optional list of arguments for the format string
113 *
114 * \return True if the operation was successful, or false otherwise, setting
115 * errno values as specified above.
116 *
117 * \b Example
118 * \code
119 * #include "pros/llemu.hpp"
120 *
121 * void initialize() {
122 * pros::lcd::initialize();
123 * pros::lcd::print(0, "My formatted text: %d!", 2);
124 * }
125 * \endcode
126 */
127 template <typename... Params>
128 bool print(std::int16_t line, const char* fmt, Params... args) {
129 return pros::c::lcd_print(line, fmt, convert_args(args)...);
130 }
