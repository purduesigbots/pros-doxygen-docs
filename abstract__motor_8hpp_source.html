102 /**
103 * Sets the voltage for the motor from -127 to 127.
104 *
105 * This is designed to map easily to the input from the controller's analog
106 * stick for simple opcontrol use. The actual behavior of the motor is
107 * analogous to use of motor_move(), or motorSet() from the PROS 2 API.
108 *
109 * This function uses the following values of errno when an error state is
110 * reached:
111 * ENODEV - The port cannot be configured as a motor
112 *
113 * \param voltage
114 * The new motor voltage from -127 to 127
115 *
116 * \return 1 if the operation was successful or PROS_ERR if the operation
117 * failed, setting errno.
118 */
119 virtual std::int32_t move(std::int32_t voltage) const = 0;
120
121 /**
122 * Sets the target absolute position for the motor to move to.
123 *
124 * This movement is relative to the position of the motor when initialized or
125 * the position when it was most recently reset with
126 * pros::Motor::set_zero_position().
127 *
128 * \note This function simply sets the target for the motor, it does not block
129 * program execution until the movement finishes.
130 *
131 * This function uses the following values of errno when an error state is
132 * reached:
133 * ENODEV - The port cannot be configured as a motor
134 *
135 * \param position
136 * The absolute position to move to in the motor's encoder units
137 * \param velocity
138 * The maximum allowable velocity for the movement in RPM
139 *
140 * \return 1 if the operation was successful or PROS_ERR if the operation
141 * failed, setting errno.
142 */
143 virtual std::int32_t move_absolute(const double position, const std::int32_t velocity) const = 0;
144
145 /**
146 * Sets the relative target position for the motor to move to.
147 *
148 * This movement is relative to the current position of the motor as given in
149 * pros::Motor::motor_get_position(). Providing 10.0 as the position parameter
150 * would result in the motor moving clockwise 10 units, no matter what the
151 * current position is.
152 *
153 * \note This function simply sets the target for the motor, it does not block
154 * program execution until the movement finishes.
155 *
156 * This function uses the following values of errno when an error state is
157 * reached:
158 * ENODEV - The port cannot be configured as a motor
159 *
160 * \param position
161 * The relative position to move to in the motor's encoder units
162 * \param velocity
163 * The maximum allowable velocity for the movement in RPM
164 *
165 * \return 1 if the operation was successful or PROS_ERR if the operation
166 * failed, setting errno.
167 */
168 virtual std::int32_t move_relative(const double position, const std::int32_t velocity) const = 0;
169
170 /**
171 * Sets the velocity for the motor.
172 *
173 * This velocity corresponds to different actual speeds depending on the
174 * gearset used for the motor. This results in a range of +-100 for
175 * E_MOTOR_GEARSET_36, +-200 for E_MOTOR_GEARSET_18, and +-600 for
176 * E_MOTOR_GEARSET_6. The velocity is held with PID to ensure consistent
177 * speed, as opposed to setting the motor's voltage.
178 *
179 * This function uses the following values of errno when an error state is
180 * reached:
181 * ENODEV - The port cannot be configured as a motor
182 *
183 * \param velocity
184 * The new motor velocity from -+-100, +-200, or +-600 depending on the
185 * motor's gearset
186 *
187 * \return 1 if the operation was successful or PROS_ERR if the operation
188 * failed, setting errno.
189 */
190 virtual std::int32_t move_velocity(const std::int32_t velocity) const = 0;
