92/**
93 * \struct vision_object_s_t
94 * This structure contains a descriptor of an object detected by the Vision Sensor
95 */
96typedef struct __attribute__((__packed__)) vision_object {
97 /// Object signature
98 uint16_t signature;
99 /// Object type, e.g. normal, color code, or line detection
100 vision_object_type_e_t type;
101 /// Left boundary coordinate of the object
102 int16_t left_coord;
103 /// Top boundary coordinate of the object
104 int16_t top_coord;
105 /// Width of the object
106 int16_t width;
107 /// Height of the object
108 int16_t height;
109 /// Angle of a color code object in 0.1 degree units (e.g. 10 -> 1 degree, 155 -> 15.5 degrees)
110 uint16_t angle;
111 /// Coordinates of the middle of the object (computed from the values above)
112 int16_t x_middle_coord;
113 /// Coordinates of the middle of the object (computed from the values above)
114 int16_t y_middle_coord;
115} vision_object_s_t;
